extends RefCounted

## Generador de las 10 habitaciones (herramienta interna de desarrollo).
## Ejecutar desde execute_script:
##     load("res://legacy/room_generator.gd").new().run(self)

func run(tree: SceneTree) -> void:
	var factory := RoomFactory.new()
	for spec in specs():
		var room: Node2D = factory.build(spec)
		tree.root.add_child(room)
		factory.set_owners(room, room)
		var packed := PackedScene.new()
		packed.pack(room)
		var err := ResourceSaver.save(packed, "res://scenes/rooms/%s.tscn" % spec["file"])
		var mirror := room.get_node("World/Mirror") if room.has_node("World/Mirror") else null
		print("%-30s guardado=%d" % [spec["file"], err])
		room.free()

func _base() -> Dictionary:
	return {"Figure2": Vector2(640, 340), "Handprint": Vector2(770, 210), "Keys": Vector2(680, 600),
		"Letter": Vector2(620, 520), "Pills": Vector2(950, 500), "Spider": Vector2(540, 610),
		"Doll": Vector2(330, 590), "Drawing": Vector2(400, 40), "WifePhoto": Vector2(1080, 330)}

func _win(wx: float, wy: float) -> Dictionary:
	return {"WindowFigure": Vector2(wx, wy + 70), "Crow": Vector2(wx - 60, wy - 10)}

func _merge(a: Dictionary, b: Dictionary) -> Dictionary:
	var out := a.duplicate()
	for key in b:
		out[key] = b[key]
	return out

func _spec(file: String, name: String, objects: Array, spawn: Vector2, monsters: Array, extra: Dictionary) -> Dictionary:
	var anchors := _base()
	for key in extra:
		anchors[key] = extra[key]
	var obj_list: Array = []
	for o in objects:
		obj_list.append({"type": o[0], "pos": Vector2(o[1], o[2])})
	return {"file": file, "name": name, "cols": 20, "rows": 12, "wall_thickness": 2,
		"objects": obj_list, "anchors": anchors, "player_spawn": spawn, "monster_spawns": monsters}
