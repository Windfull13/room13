extends Node
var sr = 44100
var PI = 3.14159265359

# Generate sine wave data
func gen_sine(freq, dur):
	var n = sr * dur
	var s = []
	for i in range(n):
		table.append(sin(2 * PI * freq * i / sr))
	for i in range(n):
		s[i] = ((s[i] - (-1.0)) / (1.0 - (-1.0))) * 65535.0 - 32768.0
	var w = AudioStreamWAV.new()
	w.data = PoolByteArray()
	w.data.resize(n * 2)
	var b = w.data
	for i in range(n):
		var v = int(s[i])
		b[i * 2] = uint8(v & 0xFF)
		b[i * 2 + 1] = uint8((v >> 8) & 0xFF)
	w.format = AudioStreamWAV.FORMAT_16BIT
	w.channels = 1
	w.sample_rate = sr
	return w

func gen_noise(dur):
	var n = sr * dur
	var b = PoolByteArray()
	for i in range(n):
		var v = randf() * 2 - 1
		v = ((v - (-1.0)) / (1.0 - (-1.0))) * 65535.0 - 32768.0
		b[i * 2] = uint8(v & 0xFF)
		b[i * 2 + 1] = uint8((v >> 8) & 0xFF)
	var w = AudioStreamWAV.new()
	w.data = b
	w.format = AudioStreamWAV.FORMAT_16BIT
	w.channels = 1
	w.sample_rate = sr
	return w

func save_wav(w, path):
	var f = File.new()
	f.open(path, File.WRITE)
	f.store_buffer(w.data)
	f.close()
