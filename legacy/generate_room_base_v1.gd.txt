## Genera la HABITACIÓN BASE reutilizable (herramienta interna de desarrollo).
##
## Ejecutar desde execute_script:
##     load("res://legacy/generate_room_base.gd").new().run(self)
##
## Las diez rondas de resources/rooms/ apuntan todas a la escena que genera
## este script. Las anomalías no crean escenas: modifican esta habitación.

func run(tree: SceneTree) -> void:
	var factory = load("res://scripts/room_factory.gd").new()
	var room: Node2D = factory.build(spec())
	tree.root.add_child(room)
	factory.set_owners(room, room)
	var packed := PackedScene.new()
	packed.pack(room)
	print("room_base.tscn -> ", ResourceSaver.save(packed, "res://scenes/rooms/room_base.tscn"))
	report(room)
	room.free()

func report(room: Node2D) -> void:
	var world := room.get_node("World")
	for name in ["Bed", "Wardrobe", "Desk", "Chair", "Window", "Door", "Clock", "Mirror", "Plant", "Toy", "Rug", "Photograph", "Phone", "Lamp", "Book"]:
		var node := world.get_node_or_null(name)
		if node == null:
			print("  FALTA ", name)
			continue
		var sprite := node as Sprite2D
		if sprite == null:
			sprite = node.get_node_or_null("Sprite") as Sprite2D
		var size: Vector2 = sprite.texture.get_size() * sprite.scale if sprite != null else Vector2.ZERO
		print("  %-11s tam=%-16s" % [name, size])
